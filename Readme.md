describe('Maurício Oliveira', () => {
  it('trabalha como QA Engineer na NTT DATA', () => {
    cy.cliente().should('eq', 'Getnet / Santander')
  })

  it('testa meios de pagamento', () => {
    cy.projetos().should('include.members', ['Pix', 'Câmbio (FX)', 'Adquirência'])
  })

  it('automatiza o que é repetitivo', () => {
    cy.cenariosCypress().should('be.gte', 50)      // rodando em Jenkins + Bitbucket
    cy.apisTestadas().should('be.gte', 100)
    cy.horasEconomizadasPorSprint().should('be.gte', 36)
  })

  it('leva qualidade para o pipeline e a cloud', () => {
    cy.qa().should('include', 'CI/CD').and('include', 'Cloud')
  })
})
